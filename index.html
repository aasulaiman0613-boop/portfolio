Example: server-side mission validationLuau
function MissionService:CompleteContract(player, contractId, payload)
    local contract = self.ActiveContracts[player]
    if not contract or contract.Id ~= contractId then return false end
    if not VehicleService:CanUseVehicle(player, payload.vehicleId) then return false end
    if not self:IsValidDropoff(contract, payload.position) then return false end
    return ProfileService:Transaction(player, "ContractReward", function(profile)
        profile.Wallet.Cash += contract.Reward.Cash
        profile.Progression.XP += contract.Reward.XP
    end)
end
